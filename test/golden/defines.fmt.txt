module name #(
    parameter WIDTH = 8
) (
    input  wire             clk,
    input  wire             rst_n,
    input  wire [WIDTH-1:0] din,
    output reg  [WIDTH-1:0] dout
);

    xpm_memory_dprom #(
        .ADDR_WIDTH_A       (6              ),
        .ADDR_WIDTH_B       (6              ),
        .AUTO_SLEEP_TIME    (0              ),
        .CASCADE_HEIGHT     (0              ),
        .CLOCKING_MODE      ("common_clock" ),  // "common_clock"/"independent_clock"
        .ECC_MODE           ("no_ecc"       ),
        .MEMORY_INIT_FILE   ("none"         ),
        .MEMORY_INIT_PARAM  ("0"            ),
        .MEMORY_OPTIMIZATION("true"         ),
        .MEMORY_PRIMITIVE   ("auto"         ),
        .MEMORY_SIZE        (2048           ),
        .MESSAGE_CONTROL    (0              ),
        .READ_DATA_WIDTH_A  (32             ),
        .READ_DATA_WIDTH_B  (32             ),
        .READ_LATENCY_A     (2              ),
        .READ_LATENCY_B     (2              ),
        .READ_RESET_VALUE_A ("0"            ),
        .READ_RESET_VALUE_B ("0"            ),
        .RST_MODE_A         ("SYNC"         ),
        .RST_MODE_B         ("SYNC"         ),
        .SIM_ASSERT_CHK     (0              ),
        .USE_MEM_INIT       (1              ),
        .USE_MEM_INIT_MMI   (0              ),
        .WAKEUP_TIME        ("disable_sleep")
    ) xpm_memory_dprom_inst (
        .dbiterr_a(dbiterr_a),  // 1 位输出：A 口 ECC 双比特错
        .dbiterr_b(dbiterr_b),  // 1 位输出：B 口 ECC 双比特错
        .douta    (douta    ),  // A 口读数据输出
        .doutb    (doutb    ),  // B 口读数据输出
        .sbiterr_a(sbiterr_a),  // 1 位输出：A 口 ECC 单比特错
        .sbiterr_b(sbiterr_b),  // 1 位输出：B 口 ECC 单比特错
        .addra    (addra    ),  // A 口读地址
        .addrb    (addrb    ),  // B 口读地址
        .clka     (clka     ),  // A 口时钟
        .clkb     (clkb     ),  // B 口时钟
        .ena      (ena      ),  // A 口使能
        .enb      (enb      ),  // B 口使能
        .injectdbiterr_a(), // A 口 ECC 双比特错误注入测试，未用悬空
        .injectdbiterr_b(), // B 口 ECC 双比特错误注入测试，未用悬空
        .injectsbiterr_a(), // A 口 ECC 单比特错误注入测试，未用悬空
        .injectsbiterr_b(), // B 口 ECC 单比特错误注入测试，未用悬空
        .regcea(1'b1),  // A 口输出寄存器使能
        .regceb(1'b1),  // B 口输出寄存器使能
        .rsta  (rst ),  // A 口复位
        .rstb  (rst ),  // B 口复位
        .sleep() // 休眠模式，WAKEUP_TIME="disable_sleep" 时未用悬空
    );
endmodule
